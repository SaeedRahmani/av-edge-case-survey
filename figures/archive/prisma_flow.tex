% ----- main column -----
\node[box] (db)  {Records identified via automated\\ database search (OpenAlex API,\\ Boolean phrase queries; through 2026-06-07)\\ \textbf{$n = 2{,}830$}};
\node[box, right=64mm of db] (expert) {Expert-identified papers\\ (263 source records; 244\\ unique papers)\\ \textbf{$n = 244$}};

\node[box, below=10mm of db] (dedup) {Records after duplicate, near-duplicate,\\ paper-owned overlap, preprint-mill,\\ and source-exclusion removal\\ \textbf{$n = 2{,}519$ new}};
\node[box, below=8mm of dedup] (screen) {Records checked by automated\\ relevance criteria (topic +\\ expert-paper similarity)\\ \textbf{$n = 2{,}519$}};
\node[box, below=8mm of screen] (elig) {Records assessed for eligibility\\ (centroid classification, ranking,\\ and human-review flagging)\\ \textbf{$n = 706$}};
\node[box, below=8mm of elig, fill=green!6] (incl) {Studies in living bibliography\\ \textbf{$n = 364$}\ \ ($244$ expert-identified $+\,120$ newly added)};

% ----- exclusion boxes -----
\node[excl, right=12mm of dedup]  (e1) {Removed: duplicate /\\ near-duplicate, already\\ paper-owned, preprint-mill,\\ or configured source exclusion\\ \textbf{$n = 311$}};
\node[excl, right=12mm of screen] (e2) {Excluded: below relevance\\ floor (off-topic or low\\ expert-paper similarity)\\ \textbf{$n = 1{,}813$}};
\node[excl, right=12mm of elig]   (e3) {Not selected in curated\\ ranking (beyond the soft\\ per-class ceiling; archived\\ in full scored list)\\ \textbf{$n = 586$}};

% ----- arrows -----
\draw[arr] (db) -- (dedup);
% Expert-identified papers feed the included set down the far right.
\draw[arr] (expert.south) -- (expert.south |- incl.east) -- (incl.east);
\draw[arr] (dedup) -- (screen);
\draw[arr] (screen) -- (elig);
\draw[arr] (elig) -- (incl);
\draw[arr] (dedup.east) -- (e1.west);
\draw[arr] (screen.east) -- (e2.west);
\draw[arr] (elig.east) -- (e3.west);

% ----- stage labels (left rail) -----
\node[stage, left=4mm of db.west, yshift=-3mm] {Identification};
\node[stage, left=4mm of screen.west] {Screening};
